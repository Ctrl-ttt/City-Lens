cmake_minimum_required(VERSION 3.20)
project(citylens_link2 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This bridge requires Windows x64")
endif()
set(LINK_SDK_ROOT "" CACHE PATH "Official Link-SDK checkout")
set(SDK "${LINK_SDK_ROOT}/UVCCamera_win/x64")
if(NOT EXISTS "${SDK}/lib/UVCCamera.lib")
  message(FATAL_ERROR "Set LINK_SDK_ROOT to the official Link-SDK checkout")
endif()
add_executable(citylens-link2 main.cpp)
target_compile_features(citylens-link2 PRIVATE cxx_std_17)
target_compile_definitions(citylens-link2 PRIVATE WIN32 NOMINMAX)
target_compile_options(citylens-link2 PRIVATE /utf-8 /W4)
target_include_directories(citylens-link2 PRIVATE "${SDK}/include")
target_link_libraries(citylens-link2 PRIVATE "${SDK}/lib/UVCCamera.lib" ole32)
add_custom_command(TARGET citylens-link2 POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
  "${SDK}/bin/UVCCamera.dll" "$<TARGET_FILE_DIR:citylens-link2>")

include(CTest)
if(BUILD_TESTING)
  add_executable(citylens-link2-ptz-test tests/ptz.cpp)
  target_compile_features(citylens-link2-ptz-test PRIVATE cxx_std_17)
  target_compile_definitions(citylens-link2-ptz-test PRIVATE WIN32 NOMINMAX)
  target_compile_options(citylens-link2-ptz-test PRIVATE /utf-8 /W4)
  target_include_directories(citylens-link2-ptz-test PRIVATE tests)
  target_link_libraries(citylens-link2-ptz-test PRIVATE ole32)
  add_test(NAME link2-ptz-axes COMMAND citylens-link2-ptz-test)
endif()
